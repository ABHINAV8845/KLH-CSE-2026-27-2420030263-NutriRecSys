def explain_prediction(features_df, raw_user=None):
    model, exp_gb, exp_rf, weights = _load_model_and_explainers()
    X = features_df[FEATURE_COLS]

    # Composite attribution weighted by meta-learner coefficients

# rag_pipeline.py — Multi-index retrieval grounded in Food.com & USDA
def execute_rag_pipeline(user_row, predicted_calories, predicted_protein, predicted_carbs, predicted_fat):
    catalog = get_catalog()
    adj_calories, adj_carbs, constraint_messages = apply_constraints(predicted_calories, predicted_carbs, user_row)
    
    # Filter recipes matching patient slot targets and clinical safety limits
    candidates = filter_candidates_by_biomarkers(catalog, user_row)
    plan = select_optimal_macro_plan(candidates, adj_calories, predicted_protein, adj_carbs, predicted_fat)
    
    return {
        "recommendations": plan["meals"],
        "nutrition": plan["nutrition"],
        "constraints": constraint_messages,
        "retrieved_context": plan["evidence"]
    }
# constraint_layer.py — Hard clinical biomarker override rules
def apply_constraints(predicted_calories, predicted_carbs_g, row):
    messages = []
    adj_carbs = predicted_carbs_g
    adj_calories = predicted_calories
    
    # Diabetic constraint: Cap carbohydrate allowance if HbA1c >= 6.5%
    if row.get("DIABETIC_FLAG", 0) == 1 or row.get("HBA1C", 0) >= 6.5:
        cap = adj_carbs * 0.75
        if adj_carbs > cap:
            messages.append(f"Carbohydrate target capped at {cap:.0f}g due to elevated HbA1c ({row.get('HBA1C'):.1f}%).")
            adj_carbs = cap
            
    return adj_calories, adj_carbs, messages
